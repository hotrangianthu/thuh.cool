Purpose
A public-facing web portal displaying Vietnam's state budget data as interactive infographics, inspired by Singapore's singaporebudget.gov.sg. Designed for public comprehension with ECharts visualizations.

4.2 js/charts.js — Available Chart Functions

// Exposed via window.VNBudgetCharts

renderBudgetPieChart(containerId, data)      // Donut chart for sector allocation
renderTrendChart(containerId, trendData)     // Line chart for revenue/expenditure trends
renderSectorBreakdownChart(containerId, data) // Horizontal bar chart for sector breakdown
renderGaugeChart(containerId, value, title)  // Semi-circle gauge for KPIs
renderRevenueTreemap(containerId, data)      // Treemap for revenue sources
initDashboardCharts()                        // Auto-init all dashboard charts
initSectorCharts(sectorId)                   // Auto-init sector page charts

6. How to Add a New Sector Page with Charts

Step 1: Add sector data to budget-2025.json

"sector_details": {
  "defense": {
    "total_budget": 275.5,
    "breakdown": [
      { "item": "Personnel", "amount": 120.0, "pct": 43.6 },
      { "item": "Equipment", "amount": 85.0, "pct": 30.8 },
      { "item": "Operations", "amount": 70.5, "pct": 25.6 }
    ],
    "kpis": {
      "defense_gdp_pct": 2.3,
      "yoy_growth_pct": 5.2
    }
  }
}

Step 2: Update sector HTML page

<!-- Add ECharts CDN in head -->
<script src="https://cdn.jsdelivr.net/npm/echarts@5.5.0/dist/echarts.min.js"></script>

<!-- Add chart container -->
<div class="sector-chart-card">
  <div class="sector-chart-title">Defense Budget Breakdown</div>
  <div id="chart-breakdown" class="chart-container-wide"></div>
</div>

<!-- Initialize chart -->
<script src="../js/charts.js"></script>
<script>
  document.addEventListener('DOMContentLoaded', () => {
    const breakdownData = {
      breakdown: [
        { item: 'Personnel', amount: 120.0 },
        { item: 'Equipment', amount: 85.0 },
        { item: 'Operations', amount: 70.5 }
      ]
    };
    window.VNBudgetCharts.renderSectorBreakdownChart('chart-breakdown', breakdownData);
  });
</script>
